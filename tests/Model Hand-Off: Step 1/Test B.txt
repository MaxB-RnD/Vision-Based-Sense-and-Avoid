import json, os
from src.introspection_module.hrnetw32_introspective.hrnetw32_introspective_test import HRNetW32IntrospectivePredictor
from src.introspection_module.hrnetw32_introspective.introspection_modules import IntrospectionModule, NullIntrospectionModule

class AlwaysEscalate(IntrospectionModule):
    name = "none"
    def should_escalate(self, track_state): return True

mode = os.environ["MODE"]
HRNetW32IntrospectivePredictor.introspection_module = AlwaysEscalate() if mode == "always" else NullIntrospectionModule()

for fl in p.get_all_flight_ids():
    res = json.load(open(os.path.join(p.get_results_directory(fl), "result.json")))
    wf = set()
    for r in res:
        for d in r["detections"]:
            det += 1
            if d["warning"]:
                warn += 1
                wf.add(r["img_name"])
    warn_frames += len(wf)
